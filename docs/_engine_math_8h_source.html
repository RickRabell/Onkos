42 inline float sqrt(float value) {
43 if (value < 0) {
44 return 0; // Handle negative input gracefully.
45 }
46 float x = value;
47 float y = 1.0f;
48 float epsilon = 0.00001f; // Desired precision.
49 while (x - y > epsilon) {
50 x = (x + y) / 2.0f;
51 y = value / x;
52 }
53 return x;
54 }

83 inline float power(float base, int exponent) {
84 if (exponent == 0) return 1;
85 if (exponent < 0) return 1.0f / power(base, -exponent);
86 float result = 1;
87 while (exponent) {
88 if (exponent % 2 == 1) result *= base;
89 base *= base;
90 exponent /= 2;
91 }
92 return result;
93 }

174 inline float sin(float angle) {
175 float result = 0.0f;
176 float term = angle;
177 float angle_squared = angle * angle;
178 int n = 1;
179 while (term > 1e-6f || term < -1e-6f) {
180 result += term;
181 term *= -angle_squared / ((2 * n) * (2 * n + 1));
182 ++n;
183 }
184 return result;
185 }

212 inline float asin(float value) {
213 // Aproximación con la serie de Taylor
214 float x = value;
215 float result = x;
216 float term = x;
217 for (int n = 1; n < 10; ++n) {
218 term *= x * x * (2 * n - 1) / ((2 * n + 1) * (2 * n));

316 inline float log(float value) {
317 // Aproximación usando la serie de Taylor
318 if (value <= 0) return 0;
319 float x = (value - 1) / (value + 1);
320 float x2 = x * x;
321 float result = 0.0f;
322 float term = x;
323 for (int n = 1; n < 20; ++n) {
324 result += term / (2 * n - 1);
325 term *= x2;
326 }
327 return 2 * result;
328 }
