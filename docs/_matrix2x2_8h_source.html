132 Matrix2x2 inverse() const {
133 float det = determinant();
134 if (det == 0) {
135 // Handle non-invertible matrix gracefully.
136 return Matrix2x2();
137 }
138 float invDet = 1.0f / det;
139 return Matrix2x2(
140 m[1][1] * invDet, -m[0][1] * invDet,
141 -m[1][0] * invDet, m[0][0] * invDet
