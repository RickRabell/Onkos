41 template<typename T>
42 class TArray
43 {
44 private:
45 T* Data;
46 size_t Capacity;
47 size_t Size;

54 void Resize(size_t NewCapacity)
55 {
56 T* NewData = new T[NewCapacity];
57 for (size_t i = 0; i < Size; ++i)
58 {
59 NewData[i] = Data[i];
60 }
61 delete[] Data;
62 Data = NewData;
63 Capacity = NewCapacity;
64 }
65
66 public:
70 TArray() : Data(nullptr), Capacity(0), Size(0) {}
71
75 ~TArray() {
76 delete[] Data;
77 }
78
84 void Add(const T& Element)
85 {
86 if (Size == Capacity)
87 {
88 Resize(Capacity == 0 ? 1 : Capacity * 2);
89 }
90 Data[Size++] = Element;
91 }
92
98 void RemoveAt(size_t Index)
99 {
100 if (Index >= Size)
101 {
102 std::cerr << "Index out of range" << std::endl;
103 return;
104 }
105 for (size_t i = Index; i < Size - 1; ++i)
106 {
107 Data[i] = Data[i + 1];
108 }
109 --Size;
110 }
111
118 T& operator[](size_t Index)
119 {
120 if (Index >= Size)
121 {
122 std::cerr << "Index out of range" << std::endl;
123 exit(1);
124 }
125 return Data[Index];
126 }
127
134 const T& operator[](size_t Index) const
135 {
136 if (Index >= Size)
137 {
138 std::cerr << "Index out of range" << std::endl;
139 exit(1);
140 }
141 return Data[Index];
142 }
143
149 size_t Num() const
150 {
151 return Size;
152 }
153
159 size_t GetCapacity() const
160 {
161 return Capacity;
162 }
163 };
