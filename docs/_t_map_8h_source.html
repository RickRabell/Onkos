41 template<typename K, typename V>
42 class TMap
43 {
44 private:
45 struct Pair
46 {
47 K Key;
48 V Value;
49
50 Pair() : Key(), Value() {}
51 Pair(const K& Key, const V& Value) : Key(Key), Value(Value) {}
52 };
53
54 Pair* Data;
55 size_t Capacity;
56 size_t Size;
57
63 void Resize(size_t NewCapacity)
64 {
65 Pair* NewData = new Pair[NewCapacity];
66 for (size_t i = 0; i < Size; ++i)
67 {
68 NewData[i] = Data[i];
69 }
70 delete[] Data;
71 Data = NewData;
72 Capacity = NewCapacity;
73 }
74
75 public:
79 TMap()
80 : Data(nullptr), Capacity(0), Size(0)
81 {
82 }
83
87 ~TMap()
88 {
89 delete[] Data;
90 }
91
98 void Add(const K& Key, const V& Value)
99 {
100 for (size_t i = 0; i < Size; ++i)
101 {
102 if (Data[i].Key == Key)
103 {
104 Data[i].Value = Value;
105 return;
106 }
107 }
108 if (Size == Capacity)
109 {
110 Resize(Capacity == 0 ? 1 : Capacity * 2);
111 }
112 Data[Size++] = Pair(Key, Value);
113 }
114
120 void Remove(const K& Key)
121 {
122 for (size_t i = 0; i < Size; ++i)
123 {
124 if (Data[i].Key == Key)
125 {
126 for (size_t j = i; j < Size - 1; ++j)
127 {
128 Data[j] = Data[j + 1];
129 }
130 --Size;
131 return;
132 }
133 }
134 std::cerr << "Key not found" << std::endl;
135 }
136
143 V& operator[](const K& Key)
144 {
145 for (size_t i = 0; i < Size; ++i)
146 {
147 if (Data[i].Key == Key)
148 {
149 return Data[i].Value;
150 }
151 }
152 std::cerr << "Key not found" << std::endl;
153 exit(1);
154 }
155
162 const V& operator[](const K& Key) const
163 {
164 for (size_t i = 0; i < Size; ++i)
165 {
166 if (Data[i].Key == Key)
167 {
168 return Data[i].Value;
169 }
170 }
171 std::cerr << "Key not found" << std::endl;
172 exit(1);
173 }
174
180 size_t Num() const
181 {
182 return Size;
183 }
184
190 size_t GetCapacity() const
191 {
192 return Capacity;
193 }
194 };
