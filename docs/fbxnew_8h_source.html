366template<typename T,typename I> T* FbxNewArray(const I n)
367{
368 const size_t lSize = FbxAllocSize(n, sizeof(T));
369 if( FBXSDK_IS_SIMPLE_TYPE(T) )
370 {
371 return (T*)FbxMalloc(lSize);
372 }
373 else
374 {
375 // malloc usually provides 8-byte or 16-byte alignment on 32bit and 64bit architectures
376 // respectively. By allocating 8 or 16 bytes for the header info, rather than sizeof(int),
377 // we ensure this function maintains the same alignment behaviour as malloc.
378 void* const pTmp = FbxMalloc(lSize + MALLOC_HEADER_SIZE);
379 *static_cast<size_t*>(pTmp) = n;
380 T* const p = reinterpret_cast<T*>(static_cast<char*>(pTmp) + MALLOC_HEADER_SIZE);
381
382 for(size_t i = 0; i < n; ++i )
383 {
384 new(p+i)T; // in-place new, not allocating memory so it is safe.
385 }
386 return p;
387 }
388}

395template<typename T> void FbxDeleteArray(T* p)
396{
397 if( p )
398 {
399 if( !FBXSDK_IS_SIMPLE_TYPE(T) )
400 {
401// When compiling on MacOS with libstdc++ we cannot use remove_const (it does not exist - not C++11)
402#ifndef USING_LIBSTDCPP
403 typedef typename std::remove_const<T>::type TMutable;
404 TMutable* const pMutable = const_cast<TMutable*>(p);
405 // FbxNewArray allocates MALLOC_HEADER_SIZE extra bytes as a header to store the array length
406 void* const pTmp = reinterpret_cast<char*>(pMutable) - MALLOC_HEADER_SIZE;
407 const int n = *static_cast<int*>(pTmp);
408#else
409 void* const pTmp = (char*)(p) - MALLOC_HEADER_SIZE;
410 const int n = *(int*)(pTmp);
411#endif
412 for( int i = 0; i < n; ++i )
413 {
414 p[i].~T();
415 }
416 FbxFree(pTmp);
417 }
418 else
419 {
420 FbxFree((void*)p);
421 }
422 }
423}
